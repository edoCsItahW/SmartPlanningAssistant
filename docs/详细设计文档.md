## 3. 核心算法模块详细设计

### 3.1 多目标优化算法实现

#### 目标函数计算
```python
class MultiObjectiveOptimizer:
    def __init__(self, user_preferences: UserPreferences):
        self.alpha = user_preferences.completion_weight  # 完成优先级权重
        self.beta = user_preferences.stability_weight    # 稳定性权重
        self.gamma = user_preferences.cognitive_weight   # 认知负荷权重
        self.delta = user_preferences.energy_weight      # 能量平衡权重
        self.epsilon = user_preferences.behavior_weight  # 行为因素权重
    
    def calculate_quality_score(self, task_sequence: List[ScheduledTask]) -> float:
        """
        计算计划质量得分：Q = α·C + β·S + γ·(1-L) + δ·E + ε·B
        """
        completion_score = self._calculate_completion_priority(task_sequence)
        stability_score = self._calculate_schedule_stability(task_sequence)
        cognitive_score = 1 - self._calculate_cognitive_load(task_sequence)
        energy_score = self._calculate_energy_balance(task_sequence)
        behavior_score = self._calculate_behavioral_factors(task_sequence)
        
        total_score = (self.alpha * completion_score +
                      self.beta * stability_score +
                      self.gamma * cognitive_score +
                      self.delta * energy_score +
                      self.epsilon * behavior_score)
        
        return total_score
    
    def _calculate_completion_priority(self, task_sequence: List[ScheduledTask]) -> float:
        """计算完成优先级得分"""
        total_score = 0.0
        total_weight = 0.0
        
        for task in task_sequence:
            # 基于紧迫性和重要性的优先级计算
            priority = (task.urgency_level * 0.6 + task.importance_level * 0.4)
            total_score += priority * task.estimated_duration
            total_weight += task.estimated_duration
        
        return total_score / total_weight if total_weight > 0 else 0.0
    
    def _calculate_cognitive_load(self, task_sequence: List[ScheduledTask]) -> float:
        """计算认知负荷，值越小越好"""
        if not task_sequence:
            return 0.0
        
        # 考虑任务类型切换的认知成本
        switch_cost = self._calculate_context_switch_cost(task_sequence)
        
        # 考虑连续高认知负荷任务的累积效应
        cumulative_load = self._calculate_cumulative_cognitive_load(task_sequence)
        
        return 0.7 * cumulative_load + 0.3 * switch_cost
    
    def _calculate_context_switch_cost(self, task_sequence: List[ScheduledTask]) -> float:
        """计算上下文切换成本"""
        if len(task_sequence) <= 1:
            return 0.0
        
        total_cost = 0.0
        for i in range(1, len(task_sequence)):
            prev_task = task_sequence[i-1]
            current_task = task_sequence[i]
            
            # 不同类型任务切换成本更高
            if prev_task.task_type != current_task.task_type:
                total_cost += 0.3
            # 高认知负荷任务间的切换成本
            elif (prev_task.cognitive_load_score > 0.7 and 
                  current_task.cognitive_load_score > 0.7):
                total_cost += 0.2
        
        return total_cost / (len(task_sequence) - 1)
```

#### 遗传算法优化器
```python
class GeneticAlgorithmOptimizer:
    def __init__(self, population_size: int = 100, generations: int = 200):
        self.population_size = population_size
        self.generations = generations
        self.mutation_rate = 0.1
        self.crossover_rate = 0.8
    
    def optimize_schedule(self, tasks: List[Task], strategy: StrategyType) -> Plan:
        """使用遗传算法优化任务安排"""
        population = self._initialize_population(tasks)
        
        for generation in range(self.generations):
            # 评估适应度
            fitness_scores = [self._evaluate_fitness(individual, strategy) 
                            for individual in population]
            
            # 选择
            selected = self._tournament_selection(population, fitness_scores)
            
            # 交叉和变异
            new_population = self._crossover_and_mutate(selected)
            
            population = new_population
        
        # 返回最佳个体
        best_index = np.argmax([self._evaluate_fitness(ind, strategy) for ind in population])
        return self._create_plan_from_individual(population[best_index], tasks)
    
    def _initialize_population(self, tasks: List[Task]) -> List[List[int]]:
        """初始化种群"""
        population = []
        for _ in range(self.population_size):
            # 随机排列任务顺序
            individual = list(range(len(tasks)))
            np.random.shuffle(individual)
            population.append(individual)
        return population
    
    def _evaluate_fitness(self, individual: List[int], strategy: StrategyType) -> float:
        """计算个体适应度"""
        scheduled_tasks = self._create_scheduled_tasks(individual)
        optimizer = MultiObjectiveOptimizer(self._get_strategy_weights(strategy))
        return optimizer.calculate_quality_score(scheduled_tasks)
```

```java
@Service
@Slf4j
public class TaskManagementService {
    
    private final TaskRepository taskRepository;
    private final TaskEvaluationService evaluationService;
    private final CacheManager cacheManager;
    
    public TaskManagementService(TaskRepository taskRepository,
                               TaskEvaluationService evaluationService,
                               CacheManager cacheManager) {
        this.taskRepository = taskRepository;
        this.evaluationService = evaluationService;
        this.cacheManager = cacheManager;
    }
    
    @Transactional
    public Task createTask(CreateTaskRequest request, String userId) {
        log.info("Creating task for user: {}", userId);
        
        // 验证任务数据
        validateTaskRequest(request);
        
        // 创建任务实体
        Task task = new Task();
        task.setUserId(userId);
        task.setTitle(request.getTitle());
        task.setDescription(request.getDescription());
        task.setUrgencyLevel(request.getUrgencyLevel());
        task.setImportanceLevel(request.getImportanceLevel());
        task.setEstimatedDuration(request.getEstimatedDuration());
        task.setTaskType(request.getTaskType());
        
        // 评估任务心理属性
        TaskEvaluationResult evaluation = evaluationService.evaluateTask(task);
        task.setCognitiveLoadScore(evaluation.getCognitiveLoadScore());
        task.setEnergyCostScore(evaluation.getEnergyCostScore());
        task.setIntrinsicMotivationScore(evaluation.getIntrinsicMotivationScore());
        
        // 保存任务
        Task savedTask = taskRepository.save(task);
        
        // 清除缓存
        cacheManager.evictUserTasksCache(userId);
        
        log.info("Task created successfully: {}", savedTask.getId());
        return savedTask;
    }
    
    @Cacheable(value = "userTasks", key = "#userId")
    public List<Task> getUserTasks(String userId, TaskFilter filter) {
        log.debug("Fetching tasks for user: {} with filter: {}", userId, filter);
        
        Specification<Task> spec = buildTaskSpecification(userId, filter);
        return taskRepository.findAll(spec, Sort.by(Sort.Direction.DESC, "createdAt"));
    }
    
    private Specification<Task> buildTaskSpecification(String userId, TaskFilter filter) {
        return (root, query, cb) -> {
            List<Predicate> predicates = new ArrayList<>();
            predicates.add(cb.equal(root.get("userId"), userId));
            
            if (filter.getUrgencyLevel() != null) {
                predicates.add(cb.greaterThanOrEqualTo(
                    root.get("urgencyLevel"), filter.getUrgencyLevel()));
            }
            
            if (filter.getTaskType() != null) {
                predicates.add(cb.equal(root.get("taskType"), filter.getTaskType()));
            }
            
            return cb.and(predicates.toArray(new Predicate[0]));
        };
    }
    
    private void validateTaskRequest(CreateTaskRequest request) {
        if (request.getTitle() == null || request.getTitle().trim().isEmpty()) {
            throw new ValidationException("Task title cannot be empty");
        }
        
        if (request.getUrgencyLevel() < 1 || request.getUrgencyLevel() > 5) {
            throw new ValidationException("Urgency level must be between 1 and 5");
        }
        
        if (request.getEstimatedDuration() <= 0) {
            throw new ValidationException("Estimated duration must be positive");
        }
    }
}
```

```java
    public Plan generateDailyPlan(String userId, LocalDate date, StrategyType strategy) {
        log.info("Generating daily plan for user: {}, date: {}, strategy: {}", 
                userId, date, strategy);
        
        // 获取用户任务
        List<Task> userTasks = taskService.getUserTasks(userId, new TaskFilter());

            throw new NoTasksException("No tasks available for planning");
```

```java
@Component
public class JwtTokenProvider {
    
    @Value("${app.jwt.secret}")
    private String jwtSecret;
    
    @Value("${app.jwt.expiration}")
    private long jwtExpiration;
    
    public String generateToken(UserPrincipal user) {
        Date now = new Date();
        Date expiryDate = new Date(now.getTime() + jwtExpiration);
        
        return Jwts.builder()
                .setSubject(user.getId())
                .setIssuedAt(now)
                .setExpiration(expiryDate)
                .signWith(SignatureAlgorithm.HS512, jwtSecret)
                .compact();
    }
    
    public String getUserIdFromToken(String token) {
        Claims claims = Jwts.parser()
                .setSigningKey(jwtSecret)
                .parseClaimsJws(token)
                .getBody();
        
        return claims.getSubject();
    }
    
    public boolean validateToken(String token) {
        try {
            Jwts.parser().setSigningKey(jwtSecret).parseClaimsJws(token);
            return true;
        } catch (Exception ex) {
            log.error("JWT token validation failed: {}", ex.getMessage());
            return false;
        }
    }
}

@Component
public class JwtAuthenticationFilter extends OncePerRequestFilter {
    
    private final JwtTokenProvider tokenProvider;
    private final UserDetailsService userDetailsService;
    
    public JwtAuthenticationFilter(JwtTokenProvider tokenProvider,
                                 UserDetailsService userDetailsService) {
        this.tokenProvider = tokenProvider;
        this.userDetailsService = userDetailsService;
    }
    
    @Override
    protected void doFilterInternal(HttpServletRequest request,
                                  HttpServletResponse response,
                                  FilterChain filterChain) throws ServletException, IOException {
        try {
            String jwt = getJwtFromRequest(request);
            
            if (StringUtils.hasText(jwt) && tokenProvider.validateToken(jwt)) {
                String userId = tokenProvider.getUserIdFromToken(jwt);
                
                UserDetails userDetails = userDetailsService.loadUserByUsername(userId);
                UsernamePasswordAuthenticationToken authentication = 
                    new UsernamePasswordAuthenticationToken(userDetails, null, userDetails.getAuthorities());
                authentication.setDetails(new WebAuthenticationDetailsSource().buildDetails(request));
                
                SecurityContextHolder.getContext().setAuthentication(authentication);
            }
        } catch (Exception ex) {
            log.error("Could not set user authentication in security context", ex);
        }
        
        filterChain.doFilter(request, response);
    }
    
    private String getJwtFromRequest(HttpServletRequest request) {
        String bearerToken = request.getHeader("Authorization");
        if (StringUtils.hasText(bearerToken) && bearerToken.startsWith("Bearer ")) {
            return bearerToken.substring(7);
        }
        return null;
    }
}
```

```java
@Service
@Slf4j
public class CacheService {
    
    private final RedisTemplate<String, Object> redisTemplate;
    private final ValueOperations<String, Object> valueOperations;
    
    public CacheService(RedisTemplate<String, Object> redisTemplate) {
        this.redisTemplate = redisTemplate;
        this.valueOperations = redisTemplate.opsForValue();
    }
    
    public void cacheUserTasks(String userId, List<Task> tasks) {
        String cacheKey = buildUserTasksKey(userId);
        try {
            valueOperations.set(cacheKey, tasks, Duration.ofMinutes(10));
            log.debug("Cached tasks for user: {}, count: {}", userId, tasks.size());
        } catch (Exception e) {
            log.error("Failed to cache user tasks for: {}", userId, e);
        }
    }
    
    @SuppressWarnings("unchecked")
    public List<Task> getCachedUserTasks(String userId) {
        String cacheKey = buildUserTasksKey(userId);
        try {
            Object cached = valueOperations.get(cacheKey);
            if (cached instanceof List) {
                log.debug("Retrieved cached tasks for user: {}", userId);
                return (List<Task>) cached;
            }
        } catch (Exception e) {
            log.error("Failed to get cached tasks for: {}", userId, e);
        }
        return null;
    }
    
    public void evictUserTasksCache(String userId) {
        String cacheKey = buildUserTasksKey(userId);
        try {
            redisTemplate.delete(cacheKey);
            log.debug("Evicted tasks cache for user: {}", userId);
        } catch (Exception e) {
            log.error("Failed to evict tasks cache for: {}", userId, e);
        }
    }
    
    private String buildUserTasksKey(String userId) {
        return "user:tasks:" + userId;
    }
}
```

```python
class OptimizedGeneticAlgorithm(GeneticAlgorithmOptimizer):
    def __init__(self, population_size=50, generations=100, early_stopping=10):
        super().__init__(population_size, generations)
        self.early_stopping = early_stopping
        self.best_fitness_history = []
    
    def optimize_schedule(self, tasks: List[Task], strategy: StrategyType) -> Plan:
        """优化版的遗传算法，支持早停和并行计算"""
        population = self._initialize_population(tasks)
        best_fitness = -float('inf')
        no_improvement_count = 0
        
        for generation in range(self.generations):
            # 并行计算适应度
            fitness_scores = self._parallel_evaluate_fitness(population, strategy, tasks)
            
            current_best = max(fitness_scores)
            self.best_fitness_history.append(current_best)
            
            # 早停检查
            if current_best > best_fitness:
                best_fitness = current_best
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            if no_improvement_count >= self.early_stopping:
                logger.info(f"Early stopping at generation {generation}")
                break
            
            # 选择、交叉、变异
            selected = self._rank_selection(population, fitness_scores)
            population = self._parallel_crossover_and_mutate(selected)
        
        best_index = np.argmax(fitness_scores)
        return self._create_plan_from_individual(population[best_index], tasks)
    
    def _parallel_evaluate_fitness(self, population, strategy, tasks):
        """并行计算种群适应度"""
        with ThreadPoolExecutor() as executor:
            futures = [executor.submit(self._evaluate_individual, ind, strategy, tasks) 
                      for ind in population]
            return [future.result() for future in futures]
    
    def _evaluate_individual(self, individual, strategy, tasks):
        """评估单个个体的适应度"""
        scheduled_tasks = self._create_scheduled_tasks(individual, tasks)
        optimizer = MultiObjectiveOptimizer(self._get_strategy_weights(strategy))
        return optimizer.calculate_quality_score(scheduled_tasks)
```

```java
        String methodName = joinPoint.getSignature().getName();
        String className = joinPoint.getTarget().getClass().getSimpleName();
        
        long startTime = System.currentTimeMillis();
        try {
            Object result = joinPoint.proceed();
            long duration = System.currentTimeMillis() - startTime;
            
            log.info("Service method completed - {}.{}: {}ms", 
                    className, methodName, duration);
                    
            // 记录指标
            meterRegistry.timer("planning.service.methods.duration", 
                    "class", className, 
                    "method", methodName)
                    .record(duration, TimeUnit.MILLISECONDS);
                    
            return result;
        } catch (Exception e) {
            log.error("Service method failed - {}.{}: {}", 
                    className, methodName, e.getMessage());
                    
            // 记录错误指标
            meterRegistry.counter("planning.service.methods.errors",
                    "class", className,
                    "method", methodName)
                    .increment();
                    
            throw e;
        }
    }
```

```java
                .timestamp(LocalDateTime.now())
                .build();

                .message(ex.getMessage())
                .timestamp(LocalDateTime.now())
```

```java
@Component
@Slf4j
public class AlgorithmServiceCircuitBreaker {
    
    private final CircuitBreakerRegistry circuitBreakerRegistry;
    private final AlgorithmService algorithmService;
    
    private static final String CIRCUIT_BREAKER_NAME = "algorithmService";
    
    public AlgorithmServiceCircuitBreaker(CircuitBreakerRegistry circuitBreakerRegistry,
                                        AlgorithmService algorithmService) {
        this.circuitBreakerRegistry = circuitBreakerRegistry;
        this.algorithmService = algorithmService;
    }
    
    public Plan generatePlanWithCircuitBreaker(List<Task> tasks, StrategyType strategy, String userId) {
        CircuitBreaker circuitBreaker = circuitBreakerRegistry.circuitBreaker(CIRCUIT_BREAKER_NAME);
        
        return circuitBreaker.executeSupplier(() -> {
            try {
                return algorithmService.generatePlan(tasks, strategy, userId);
            } catch (Exception e) {
                log.warn("Algorithm service call failed, using fallback", e);
                throw new CallNotPermittedException("Algorithm service unavailable");
            }
        }, throwable -> {
            // Fallback: 使用简单的时间顺序安排
            log.info("Using fallback plan generation");
            return generateFallbackPlan(tasks, userId);
        });
    }
    
    private Plan generateFallbackPlan(List<Task> tasks, String userId) {
        // 简单的按创建时间排序的计划
        List<Task> sortedTasks = tasks.stream()
                .sorted(Comparator.comparing(Task::getCreatedAt))
                .collect(Collectors.toList());
                
        Plan fallbackPlan = new Plan();
        fallbackPlan.setUserId(userId);
        fallbackPlan.setPlanDate(LocalDate.now());
        fallbackPlan.setStrategyType(StrategyType.EFFICIENCY);
        fallbackPlan.setQualityScore(0.5); // 中等质量分数
        
        // 创建简单的时间安排
        List<ScheduledTask> scheduledTasks = new ArrayList<>();
        LocalTime startTime = LocalTime.of(9, 0);
        
        for (Task task : sortedTasks) {
            ScheduledTask scheduledTask = new ScheduledTask();
            scheduledTask.setTaskId(task.getId());
            scheduledTask.setScheduledStart(startTime);
            scheduledTask.setScheduledEnd(startTime.plusMinutes(task.getEstimatedDuration()));
            scheduledTasks.add(scheduledTask);
            
            startTime = scheduledTask.getScheduledEnd().plusMinutes(15); // 添加休息时间
        }
        
        fallbackPlan.setTasksSchedule(scheduledTasks);
        return fallbackPlan;
    }
}
```
